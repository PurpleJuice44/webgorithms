# 정수론 입문 - 약수, 배수, 소수와 체

void find_divisors(int n) {
    printf("%d의 약수: ", n);
    for (int i = 1; i <= n; i++) {
        if (n % i == 0) {
            printf("%d ", i);
        }
    }
    printf("\n");
}

// 예시
int main() {
    int number = 12;
    find_divisors(number);
    return 0;
}

출력: 12의 약수: 1 2 3 4 6 12

🔍 2. 배수 구하기

# include <stdio.h>

void find_multiples(int n, int count) {
    printf("%d의 배수(%d개): ", n, count);
    for (int i = 1; i <= count; i++) {
        printf("%d ", n * i);
    }
    printf("\n");
}

// 예시
int main() {
    int number = 5;
    find_multiples(number, 10);
    return 0;
}

출력: 5의 배수(10개): 5 10 15 20 25 30 35 40 45 50

🔍 3. 소수 확인하기

# include <stdio.h>
# include <math.h>

int is_prime(int n) {
    if (n < 2) return 0;
    for (int i = 2; i <= sqrt(n); i++) {
        if (n % i == 0) return 0;
    }
    return 1;
}

// 예시
int main() {
    int num = 17;
    if (is_prime(num)) {
        printf("%d은 소수입니다.\n", num);
    } else {
        printf("%d은 소수 아님.\n", num);
    }
    return 0;
}

출력: 17은 소수입니다.

보충: Python 구현과 복잡도 증명

Python - Divisors, Prime Check, Sieve

import sys, math

def divisors(n):
    res=[]
    for i in range(1, int(math.isqrt(n))+1):
        if n%i==0:
            res.append(i)
            if i*i!=n: res.append(n//i)
    return sorted(res)

def is_prime(n):
    if n<2: return False
    for i in range(2, int(math.isqrt(n))+1):
        if n%i==0: return False
    return True

def sieve(n):
    is_prime=[True]*(n+1)
    is_prime[0]=is_prime[1]=False
    for i in range(2, int(n**0.5)+1):
        if is_prime[i]:
            step=i; start=i*i
            is_prime[start:n+1:step]=[False]*(((n - start)//step)+1)
    return [i for i, v in enumerate(is_prime) if v]

def solve():
    import sys
    data=list(map(int, sys.stdin.read().split()))
    if not data: return
    n=data[0]
    print(divisors(n)); print(is_prime(n)); print(sieve(min(n,100)))
if __name__=="__main__":
    solve()

시간/공간 보충: 약수 O(sqrt n), 소수 O(sqrt n), 체 O(n log log n) 시간 O(n) 공간.
